cmake_minimum_required(VERSION 3.16)
project(mega_attn_910b LANGUAGES CXX)
option(MEGA_ASCEND_CPU_ONLY "Only validate the CPU oracle without CANN" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(bench_megaAttn bench_megaAttn.cpp)
target_compile_features(bench_megaAttn PRIVATE cxx_std_20)
target_compile_options(bench_megaAttn PRIVATE -O2 -Wall -Wextra)
if(MEGA_ASCEND_CPU_ONLY)
    target_compile_definitions(bench_megaAttn PRIVATE MEGA_ASCEND_CPU_ONLY)
    enable_testing()
    add_test(NAME mega_attn_cpu_oracle COMMAND bench_megaAttn)
    return()
endif()

set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN installation")
if(NOT ASCEND_CANN_PACKAGE_PATH)
    set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/cann-9.0.0")
endif()
set(SOC_VERSION "Ascend910B2C" CACHE STRING "Target SoC")
set(RUN_MODE "npu" CACHE STRING "Ascend execution mode")
# project() 可能已创建空缓存；CANN 合并脚本要求 build-type 非空。
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
# CANN 安装包的 CMake 模块负责设备编译和 <<<>>> 启动封装。
foreach(relative tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake)
    if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/${relative}/ascendc.cmake")
        set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/${relative}")
        break()
    endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
    message(FATAL_ERROR "Cannot find ascendc.cmake under ${ASCEND_CANN_PACKAGE_PATH}; check Toolkit installation")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
# 每个版本独立生成混合核元数据，避免 CANN 9.0 的同源多入口编号问题。
ascendc_library(mega_attn_kernel STATIC megaAttn_v1.cpp)
ascendc_library(mega_attn_kernel_v2 STATIC megaAttn_v2.cpp)
# 让启动封装初始化 Matmul 通信区并通知 AIV；workspace 必须是核函数最后一个参数。
ascendc_compile_definitions(mega_attn_kernel PRIVATE HAVE_WORKSPACE)
ascendc_compile_definitions(mega_attn_kernel_v2 PRIVATE HAVE_WORKSPACE)
target_include_directories(bench_megaAttn PRIVATE
    "${ASCEND_CANN_PACKAGE_PATH}/include"
    "${ASCEND_CANN_PACKAGE_PATH}/include/experiment")
target_link_directories(bench_megaAttn PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_compile_definitions(bench_megaAttn PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
target_link_libraries(bench_megaAttn PRIVATE mega_attn_kernel mega_attn_kernel_v2 ascendcl
    tiling_api platform register graph_base ascendalog dl)
set_target_properties(bench_megaAttn PROPERTIES BUILD_RPATH "${ASCEND_CANN_PACKAGE_PATH}/lib64")
